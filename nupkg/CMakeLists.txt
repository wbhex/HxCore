configure_file(${CMAKE_CURRENT_SOURCE_DIR}/nupkg.nuspec.in ${CMAKE_CURRENT_BINARY_DIR}/${PROJECT_NAME}.nuspec @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/${PROJECT_NAME}.nuspec DESTINATION ${HX_INSTALL_DIR})

set(HX_NUPKG_TARGET_NAME ${PROJECT_NAME}.${HX_ARCHITECTURE}.${HX_SYSTEM})

configure_file(${CMAKE_CURRENT_SOURCE_DIR}/nupkg.props.in ${CMAKE_CURRENT_BINARY_DIR}/${HX_NUPKG_TARGET_NAME}.props @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/${HX_NUPKG_TARGET_NAME}.props DESTINATION ${HX_INSTALL_DIR}/build)

configure_file(${CMAKE_CURRENT_SOURCE_DIR}/nupkg.targets.in ${CMAKE_CURRENT_BINARY_DIR}/${HX_NUPKG_TARGET_NAME}.targets @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/${HX_NUPKG_TARGET_NAME}.targets DESTINATION ${HX_INSTALL_DIR}/build)

if(CMAKE_SYSTEM_NAME MATCHES "Windows")
	configure_file(${CMAKE_CURRENT_SOURCE_DIR}/nupkgPack.bat.in ${CMAKE_SOURCE_DIR}/install/${HX_NUPKG_TARGET_NAME}.nupkgPack.bat @ONLY)
else()
	configure_file(${CMAKE_CURRENT_SOURCE_DIR}/nupkgPack.sh.in ${CMAKE_SOURCE_DIR}/install/${HX_NUPKG_TARGET_NAME}.nupkgPack.sh @ONLY)
endif()

configure_file(${CMAKE_CURRENT_SOURCE_DIR}/nupkg.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/${PROJECT_NAME}Config.cmake @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/${PROJECT_NAME}Config.cmake DESTINATION ${HX_INSTALL_DIR}/build/native/cmake)

install(DIRECTORY ${CMAKE_SOURCE_DIR}/include DESTINATION ${HX_INSTALL_DIR}/build/native/)
