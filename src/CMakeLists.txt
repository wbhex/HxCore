set(TARGET_NAME ${PROJECT_NAME})

set(CMAKE_CXX_STANDARD 17)

add_definitions(-DHX_NOEXCEPT_ENABLED)
add_definitions(-DHX_WORKING_SPECIFIED)

file(GLOB_RECURSE HEADER_FILES *.h *.hpp)
file(GLOB_RECURSE SOURCE_FILES *.c *.cpp)

if(CMAKE_SYSTEM_NAME MATCHES "Windows")

	file(GLOB_RECURSE RESOURCE_FILES "./*.rc")

	configure_file(${CMAKE_SOURCE_DIR}/resource.rc.in ${CMAKE_CURRENT_BINARY_DIR}/${TARGET_NAME}.rc @ONLY)
	configure_file(${CMAKE_SOURCE_DIR}/resource.h.in ${CMAKE_CURRENT_BINARY_DIR}/resource.h @ONLY)
	list(APPEND RESOURCE_FILES ${CMAKE_CURRENT_BINARY_DIR}/${TARGET_NAME}.rc)
	list(APPEND RESOURCE_FILES ${CMAKE_CURRENT_BINARY_DIR}/resource.h)

	add_library(${TARGET_NAME} SHARED ${HEADER_FILES} ${SOURCE_FILES} ${RESOURCE_FILES})
else()
	add_library(${TARGET_NAME} SHARED ${HEADER_FILES} ${SOURCE_FILES})
	target_link_libraries(${TARGET_NAME} PRIVATE atomic)
	target_link_libraries(${TARGET_NAME} PRIVATE pthread)
	target_link_libraries(${TARGET_NAME} PRIVATE ${CMAKE_DL_LIBS})
	set_target_properties(${TARGET_NAME} PROPERTIES VERSION ${PROJECT_VERSION_MAJOR}.${PROJECT_VERSION_MINOR}.${PROJECT_VERSION_PATCH}.${HX_VERSION_TAG} SOVERSION ${PROJECT_VERSION_MAJOR})
endif()

#安装时处理
install(TARGETS ${TARGET_NAME}
	RUNTIME DESTINATION ${HX_INSTALL_RUNTIME_FOLDER}
	LIBRARY DESTINATION ${HX_INSTALL_LIBRARY_FOLDER}
	ARCHIVE DESTINATION ${HX_INSTALL_ARCHIVE_FOLDER})
