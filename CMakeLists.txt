cmake_minimum_required(VERSION 3.12)
project(HxCore VERSION 1.0.0.0)

#统一初始化
include(${CMAKE_SOURCE_DIR}/cmake/Init.cmake)
HxInit()
#获取版本标签
include(${CMAKE_SOURCE_DIR}/cmake/VersionTag.cmake)
HxGetVersionTag()

#设置安装目录
set(HX_INSTALL_DIR "${PROJECT_SOURCE_DIR}/install/${HX_SYSTEM}-${HX_ARCHITECTURE}-${CMAKE_BUILD_TYPE}")

#
if(CMAKE_SYSTEM_NAME MATCHES "Windows")
	set(HX_TARGET_EXTENSION "dll")
	set(HX_LIBNAME_PREFIX "")
	set(HX_LIBNAME_SUFFIX ".dll")
	set(HX_FILEPATH_SEPARATOR "\\")
	#设置安装路径
	SET(HX_INSTALL_RUNTIME_FOLDER "${HX_INSTALL_DIR}/build/native/bin")
	SET(HX_INSTALL_LIBRARY_FOLDER "${HX_INSTALL_DIR}/build/native/lib")
	SET(HX_INSTALL_ARCHIVE_FOLDER "${HX_INSTALL_DIR}/build/native/lib")
else()
	set(HX_TARGET_EXTENSION "so")
	set(HX_LIBNAME_PREFIX "lib")
	set(HX_LIBNAME_SUFFIX ".so")
	set(HX_FILEPATH_SEPARATOR "/")
	#设置依赖默认路径为当前目录
	set(CMAKE_BUILD_RPATH "$ORIGIN")
	set(CMAKE_INSTALL_RPATH "$ORIGIN")
	#设置安装路径
	SET(HX_INSTALL_RUNTIME_FOLDER "${HX_INSTALL_DIR}/build/native/bin")
	SET(HX_INSTALL_LIBRARY_FOLDER "${HX_INSTALL_DIR}/build/native/bin")
	SET(HX_INSTALL_ARCHIVE_FOLDER "${HX_INSTALL_DIR}/build/native/lib")
endif()

include_directories(${CMAKE_SOURCE_DIR}/include/)

#设置生成路径
if(NOT DEFINED HX_OUT_DIR)
	set(HX_OUT_DIR "${PROJECT_SOURCE_DIR}/out/${HX_SYSTEM}-${HX_ARCHITECTURE}-${CMAKE_BUILD_TYPE}")
endif()
message(STATUS "HX_OUT_DIR: ${HX_OUT_DIR}")

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${HX_OUT_DIR}")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${HX_OUT_DIR}")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${HX_OUT_DIR}")
set(CMAKE_COMPILE_PDB_OUTPUT_DIRECTORY "${HX_OUT_DIR}")
set(CMAKE_PDB_OUTPUT_DIRECTORY "${HX_OUT_DIR}")

foreach(OUTPUTCONFIG ${CMAKE_CONFIGURATION_TYPES})
    string(TOUPPER ${OUTPUTCONFIG} OUTPUTCONFIG)    
	set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${OUTPUTCONFIG} ${HX_OUT_DIR})
    set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_${OUTPUTCONFIG} ${HX_OUT_DIR})
    set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_${OUTPUTCONFIG} ${HX_OUT_DIR})
	set(CMAKE_COMPILE_PDB_OUTPUT_DIRECTORY_${OUTPUTCONFIG} "${HX_OUT_DIR}")
	set(CMAKE_PDB_OUTPUT_DIRECTORY_${OUTPUTCONFIG} "${HX_OUT_DIR}")
endforeach(OUTPUTCONFIG CMAKE_CONFIGURATION_TYPES)

#设置工作目录
set(CMAKE_WORKING_DIRECTORY "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}")
message(STATUS "CMAKE_WORKING_DIRECTORY: ${CMAKE_WORKING_DIRECTORY}")

# 包含子项目。
add_subdirectory(src)

option(HX_ENABLE_TEST_PROJECTS "Enable/disable test projects" ON)
if(HX_ENABLE_TEST_PROJECTS)
	add_subdirectory(tests)
	message(STATUS "enable test projects!")
endif()

if(CMAKE_BUILD_TYPE MATCHES "Release")
	add_subdirectory("nupkg")
endif()
